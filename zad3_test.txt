#include "xc.h"
#include "libpic30.h"
#include <stdint.h>

// ===== ADC =====
void ADC_Init() {
    AD1CON1 = 0x00E0;   // auto-convert
    AD1CON2 = 0x0000;
    AD1CON3 = 0x1F02;
    AD1CHS = 0x0000;    // AN0 (P1)
    AD1PCFG = 0xFFFE;   // AN0 analog, reszta cyfrowe
    AD1CON1bits.ADON = 1;
}

uint16_t ADC_Read() {
    AD1CON1bits.SAMP = 1;
    __delay32(1000);
    AD1CON1bits.SAMP = 0;
    while(!AD1CON1bits.DONE);
    return ADC1BUF0;
}

int main(void) {

    // LED
    TRISA = 0x0000;
    LATA = 0x0000;

    // Przycisk RD6
    TRISDbits.TRISD6 = 1;

    ADC_Init();

    uint16_t adc_val;
    uint16_t prog = 512; // połowa zakresu (10-bit)

    int alarm = 0;
    int miganie = 0;

    while(1) {

        adc_val = ADC_Read();

        // ===== WŁĄCZENIE ALARMU =====
        if(adc_val > prog && alarm == 0) {
            alarm = 1;
            miganie = 1;

            // 5 sekund migania
            for(int i = 0; i < 10; i++) {
                LATA = 0x01;
                __delay32(500000);
                LATA = 0x00;
                __delay32(500000);

                // jeśli spadnie poniżej progu -> wyłącz
                if(ADC_Read() < prog) {
                    alarm = 0;
                    miganie = 0;
                    LATA = 0x00;
                    break;
                }

                // przycisk RD6 -> wyłącz
                if(PORTDbits.RD6 == 0) {
                    alarm = 0;
                    miganie = 0;
                    LATA = 0x00;
                    break;
                }
            }

            // po miganiu – wszystkie diody
            if(alarm == 1) {
                miganie = 0;
                LATA = 0xFF;
            }
        }

        // ===== WYŁĄCZENIE POTENCJOMETREM =====
        if(alarm == 1 && adc_val < prog) {
            alarm = 0;
            LATA = 0x00;
        }

        // ===== WYŁĄCZENIE PRZYCISKIEM RD6 =====
        if(PORTDbits.RD6 == 0) {
            alarm = 0;
            LATA = 0x00;
        }

        __delay32(50000);
    }

    return 0;
}
